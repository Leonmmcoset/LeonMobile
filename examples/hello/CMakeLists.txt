cmake_minimum_required(VERSION 3.20)
project(LeonHello LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(LEON_HELLO_PKGCONFIG "Use the pkg-config SDK interface" OFF)
if(LEON_HELLO_PKGCONFIG)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LMUI REQUIRED IMPORTED_TARGET lmui)
else()
  find_package(LMUI 1 CONFIG REQUIRED)
endif()
add_executable(hello main.cpp)
if(LEON_HELLO_PKGCONFIG)
  target_link_libraries(hello PRIVATE PkgConfig::LMUI)
else()
  target_link_libraries(hello PRIVATE LMUI::LMUI)
endif()
