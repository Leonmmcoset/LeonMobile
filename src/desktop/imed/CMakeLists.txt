find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(IME_XML ${PROJECT_SOURCE_DIR}/protocols/input-method-unstable-v2.xml)
add_custom_command(OUTPUT input-method-client-protocol.h COMMAND ${WAYLAND_SCANNER} client-header ${IME_XML} input-method-client-protocol.h DEPENDS ${IME_XML})
add_custom_command(OUTPUT input-method-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${IME_XML} input-method-protocol.c DEPENDS ${IME_XML})
add_executable(imed main.cpp backend.c engine.c input-method-protocol.c input-method-client-protocol.h)
target_include_directories(imed PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(imed PRIVATE _GNU_SOURCE)
pkg_check_modules(IME_WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
target_link_libraries(imed PRIVATE leon-app-support PkgConfig::IME_WAYLAND)
install(TARGETS imed RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Runtime)
