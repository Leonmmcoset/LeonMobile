cmake_minimum_required(VERSION 3.20)
project(LeonMobile VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CTest)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
option(LEON_GRAPHICS "Build the Wayland desktop and LMUI applications" ON)
option(LEON_SYSTEM "Build init and system services" ON)
option(LEON_TOOLS "Build service and package management tools" ON)
option(LEON_LMUI "Build LMUI; otherwise consume the installed SDK" ON)
option(LEON_APPS "Build the independent built-in applications" ON)
option(LEON_SHELL "Build the mobile shell" ON)
option(LEON_COMPOSITOR "Build the wlroots compositor" ON)
add_compile_options(-Wall -Wextra -Wformat=2 $<$<COMPILE_LANGUAGE:C>:-Werror=implicit-function-declaration>)
if(LEON_GRAPHICS)
  find_package(PkgConfig REQUIRED)
endif()
add_subdirectory(src/libraries)
if(LEON_SYSTEM OR LEON_TOOLS)
  add_subdirectory(src/services/common)
endif()
if(LEON_SYSTEM)
  add_subdirectory(src/services)
endif()
if(LEON_TOOLS)
  add_subdirectory(src/tools)
endif()
if(LEON_GRAPHICS)
  if(LEON_APPS)
    add_subdirectory(src/apps)
  endif()
  add_subdirectory(src/desktop)
endif()
if(LEON_SYSTEM)
  install(FILES VERSION DESTINATION ${CMAKE_INSTALL_DATADIR}/leon RENAME version COMPONENT Runtime)
endif()
